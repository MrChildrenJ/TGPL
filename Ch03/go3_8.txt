func main() {
	const (
		xmin, ymin, xmax, ymax = -2, -2, +2, +2
		width, height          = 512, 512
	)

	fmt.Println("Mandelbrot Set with Different Precision Types")
	fmt.Println("============================================")

	// Test different zoom levels
	zoomLevels := []struct {
		name string
		xmin, ymin, xmax, ymax float64
	}{
		{"Standard View", -2, -2, +2, +2},
		{"Zoom 10x", -0.8, -0.8, -0.6, -0.6},
		{"Zoom 100x", -0.75, -0.11, -0.73, -0.09},
		{"Zoom 1000x", -0.742, -0.112, -0.740, -0.110},
	}

	for _, zoom := range zoomLevels {
		fmt.Printf("\n%s (%.6f, %.6f) to (%.6f, %.6f)\n", zoom.name, zoom.xmin, zoom.ymin, zoom.xmax, zoom.ymax)
		fmt.Println(strings.Repeat("-", 60))

		// Test complex64
		testPrecision("complex64", width, height, zoom.xmin, zoom.ymin, zoom.xmax, zoom.ymax, generateComplex64)
		
		// Test complex128  
		testPrecision("complex128", width, height, zoom.xmin, zoom.ymin, zoom.xmax, zoom.ymax, generateComplex128)
		
		// Test big.Float
		testPrecision("big.Float", width, height, zoom.xmin, zoom.ymin, zoom.xmax, zoom.ymax, generateBigFloat)
		
		// Test big.Rat
		testPrecision("big.Rat", width, height, zoom.xmin, zoom.ymin, zoom.xmax, zoom.ymax, generateBigRat)
	}
}

func testPrecision(name string, width, height int, xmin, ymin, xmax, ymax float64, generator func(int, int, float64, float64, float64, float64) *image.RGBA) {
	var m1, m2 runtime.MemStats
	runtime.GC()
	runtime.ReadMemStats(&m1)
	
	start := time.Now()
	img := generator(width, height, xmin, ymin, xmax, ymax)
	duration := time.Since(start)
	
	runtime.ReadMemStats(&m2)
	memUsed := m2.TotalAlloc - m1.TotalAlloc
	
	fmt.Printf("%-12s: %8s, Memory: %6d KB\n", name, duration.Round(time.Millisecond), memUsed/1024)
	
	// Save image
	filename := fmt.Sprintf("mandelbrot_%s.png", name)
	file, _ := os.Create(filename)
	png.Encode(file, img)
	file.Close()
}

// complex64 implementation (32-bit precision)
func generateComplex64(width, height int, xmin, ymin, xmax, ymax float64) *image.RGBA {
	img := image.NewRGBA(image.Rect(0, 0, width, height))
	for py := 0; py < height; py++ {
		y := float32(float64(py)/float64(height)*(ymax-ymin) + ymin)
		for px := 0; px < width; px++ {
			x := float32(float64(px)/float64(width)*(xmax-xmin) + xmin)
			z := complex(x, y)
			img.Set(px, py, mandelbrot64(z))
		}
	}
	return img
}

func mandelbrot64(z complex64) color.Color {
	const iterations = 200
	const contrast = 15

	var v complex64
	for n := uint8(0); n < iterations; n++ {
		v = v*v + z
		if real(v)*real(v)+imag(v)*imag(v) > 4 {
			r := uint8((n * contrast) % 255)
			g := uint8((n * contrast * 3) % 255)
			b := uint8((n * contrast * 6) % 255)
			return color.RGBA{r, g, b, 255}
		}
	}
	return color.Black
}

// complex128 implementation (64-bit precision) 
func generateComplex128(width, height int, xmin, ymin, xmax, ymax float64) *image.RGBA {
	img := image.NewRGBA(image.Rect(0, 0, width, height))
	for py := 0; py < height; py++ {
		y := float64(py)/float64(height)*(ymax-ymin) + ymin
		for px := 0; px < width; px++ {
			x := float64(px)/float64(width)*(xmax-xmin) + xmin
			z := complex(x, y)
			img.Set(px, py, mandelbrot128(z))
		}
	}
	return img
}

func mandelbrot128(z complex128) color.Color {
	const iterations = 200
	const contrast = 15

	var v complex128
	for n := uint8(0); n < iterations; n++ {
		v = v*v + z
		if cmplx.Abs(v) > 2 {
			r := uint8((n * contrast) % 255)
			g := uint8((n * contrast * 3) % 255)
			b := uint8((n * contrast * 6) % 255)
			return color.RGBA{r, g, b, 255}
		}
	}
	return color.Black
}

// big.Float implementation (arbitrary precision floating point)
func generateBigFloat(width, height int, xmin, ymin, xmax, ymax float64) *image.RGBA {
	img := image.NewRGBA(image.Rect(0, 0, width, height))
	
	// Set precision to 128 bits (much higher than float64's 53 bits)
	precision := uint(128)
	
	bXmin := big.NewFloat(xmin).SetPrec(precision)
	bYmin := big.NewFloat(ymin).SetPrec(precision)
	bXmax := big.NewFloat(xmax).SetPrec(precision)
	bYmax := big.NewFloat(ymax).SetPrec(precision)
	bWidth := big.NewFloat(float64(width)).SetPrec(precision)
	bHeight := big.NewFloat(float64(height)).SetPrec(precision)
	
	xRange := new(big.Float).SetPrec(precision).Sub(bXmax, bXmin)
	yRange := new(big.Float).SetPrec(precision).Sub(bYmax, bYmin)
	
	for py := 0; py < height; py++ {
		bPy := big.NewFloat(float64(py)).SetPrec(precision)
		y := new(big.Float).SetPrec(precision)
		y.Quo(bPy, bHeight).Mul(y, yRange).Add(y, bYmin)
		
		for px := 0; px < width; px++ {
			bPx := big.NewFloat(float64(px)).SetPrec(precision)
			x := new(big.Float).SetPrec(precision)
			x.Quo(bPx, bWidth).Mul(x, xRange).Add(x, bXmin)
			
			img.Set(px, py, mandelbrotBigFloat(x, y, precision))
		}
	}
	return img
}

func mandelbrotBigFloat(x, y *big.Float, precision uint) color.Color {
	const iterations = 200
	const contrast = 15

	vReal := big.NewFloat(0).SetPrec(precision)
	vImag := big.NewFloat(0).SetPrec(precision)
	two := big.NewFloat(2).SetPrec(precision)
	temp := big.NewFloat(0).SetPrec(precision)

	for n := uint8(0); n < iterations; n++ {
		// v = v*v + z
		// (a+bi)^2 = a^2 - b^2 + 2abi
		newReal := new(big.Float).SetPrec(precision)
		newImag := new(big.Float).SetPrec(precision)
		
		// newReal = vReal^2 - vImag^2 + x
		newReal.Mul(vReal, vReal).Sub(newReal, temp.Mul(vImag, vImag)).Add(newReal, x)
		// newImag = 2*vReal*vImag + y  
		newImag.Mul(vReal, vImag).Mul(newImag, two).Add(newImag, y)
		
		vReal.Set(newReal)
		vImag.Set(newImag)
		
		// Check if |v| > 2
		magnitude := new(big.Float).SetPrec(precision)
		magnitude.Mul(vReal, vReal).Add(magnitude, temp.Mul(vImag, vImag))
		
		if magnitude.Cmp(temp.Mul(two, two)) > 0 {
			r := uint8((n * contrast) % 255)
			g := uint8((n * contrast * 3) % 255)
			b := uint8((n * contrast * 6) % 255)
			return color.RGBA{r, g, b, 255}
		}
	}
	return color.Black
}

// big.Rat implementation (arbitrary precision rational numbers)
func generateBigRat(width, height int, xmin, ymin, xmax, ymax float64) *image.RGBA {
	img := image.NewRGBA(image.Rect(0, 0, width, height))
	
	// Convert floats to rationals with high precision denominators
	denominator := int64(1000000) // 6 decimal places precision
	
	rXmin := big.NewRat(int64(xmin*float64(denominator)), denominator)
	rYmin := big.NewRat(int64(ymin*float64(denominator)), denominator)
	rXmax := big.NewRat(int64(xmax*float64(denominator)), denominator)
	rYmax := big.NewRat(int64(ymax*float64(denominator)), denominator)
	rWidth := big.NewRat(int64(width), 1)
	rHeight := big.NewRat(int64(height), 1)
	
	xRange := new(big.Rat).Sub(rXmax, rXmin)
	yRange := new(big.Rat).Sub(rYmax, rYmin)
	
	for py := 0; py < height; py++ {
		rPy := big.NewRat(int64(py), 1)
		y := new(big.Rat)
		y.Quo(rPy, rHeight).Mul(y, yRange).Add(y, rYmin)
		
		for px := 0; px < width; px++ {
			rPx := big.NewRat(int64(px), 1)
			x := new(big.Rat)
			x.Quo(rPx, rWidth).Mul(x, xRange).Add(x, rXmin)
			
			img.Set(px, py, mandelbrotBigRat(x, y))
		}
	}
	return img
}

func mandelbrotBigRat(x, y *big.Rat) color.Color {
	const iterations = 200
	const contrast = 15

	vReal := big.NewRat(0, 1)
	vImag := big.NewRat(0, 1)
	two := big.NewRat(2, 1)
	four := big.NewRat(4, 1)
	temp := big.NewRat(0, 1)

	for n := uint8(0); n < iterations; n++ {
		// v = v*v + z
		newReal := new(big.Rat)
		newImag := new(big.Rat)
		
		// newReal = vReal^2 - vImag^2 + x
		newReal.Mul(vReal, vReal).Sub(newReal, temp.Mul(vImag, vImag)).Add(newReal, x)
		// newImag = 2*vReal*vImag + y
		newImag.Mul(vReal, vImag).Mul(newImag, two).Add(newImag, y)
		
		vReal.Set(newReal)
		vImag.Set(newImag)
		
		// Check if |v|^2 > 4
		magnitudeSquared := new(big.Rat)
		magnitudeSquared.Mul(vReal, vReal).Add(magnitudeSquared, temp.Mul(vImag, vImag))
		
		if magnitudeSquared.Cmp(four) > 0 {
			r := uint8((n * contrast) % 255)
			g := uint8((n * contrast * 3) % 255)
			b := uint8((n * contrast * 6) % 255)
			return color.RGBA{r, g, b, 255}
		}
	}
	return color.Black
}
