package main

import (
	"fmt"
	"math"
)

const epsilon = 1e-15

type ErrNegativeSqrt float64

func (e ErrNegativeSqrt) Error() string {
	return fmt.Sprintf("cannot Sqrt negative number: %v", float64(e))
}

func Sqrt(x float64) (float64, error) {
	if x < 0 {
		return math.NaN(), ErrNegativeSqrt(x) // struct use {}, primitive use ()
	}

	z := x / 2
	count := 0
	for math.Abs(z*z-x) > epsilon {
		z -= (z*z - x) / (2 * z)
		count += 1
		if count > 50 {
			break
		}
	}
	fmt.Printf("There are %v iterations\n", count)
	return z, nil
}

func main() {
	fmt.Println(Sqrt(2))
	fmt.Println(Sqrt(-2))
}
